EN

webui.ofs
attach {webui}

core main() {
    forge body = webui.nav("OFS", webui.button("Docs", "#docs"))
    body = body + webui.grid(webui.card("ODL", "Documents") + webui.card("OES", "Effects"))
    webui.serve(8080, "OFS Web", body)
}
